            d,_=b.marginal(e.position,max(0.0,e.means[0]));hard_n+=1
            if d<=1e-12:hard_zero_n+=1
        for e in targets:
            shap=phi_by_h[0][e.player_id][0];hard=hard_by_h[0][e.player_id][0];probs_by={}
            for h in (1,2):
                pr=explicit.state_distribution_for_example(e,boundaries,transitions,qb_probs,h);probs_by[h]=pr
                vals=phi_by_h[h][e.player_id];hvals=hard_by_h[h][e.player_id]
                shap += (DISCOUNT**h)*sum(pr[st]*vals[1+STATE_NAMES.index(st)] for st in STATE_NAMES)
                hard += (DISCOUNT**h)*sum(pr[st]*hvals[1+STATE_NAMES.index(st)] for st in STATE_NAMES)
            base=keys[(season,e.player_id)]
            out_rows.append({"season":season,"player_id":e.player_id,"position":e.position,"age":e.age,"experience":e.experience,
                             "target":float(base["target"]),"A":float(base["A"]),"D":float(base["D"]),"Hard":hard,"Shapley":shap,
                             "phi_y1":phi_by_h[0][e.player_id][0],"hard_y1":hard_by_h[0][e.player_id][0],
                             "state_probs_y2":probs_by[1],"state_probs_y3":probs_by[2]})
    ms={k:metrics(out_rows,k) for k in ("A","D","Hard","Shapley")}
    folds=sorted({r["season"] for r in out_rows})
    wins=sum(ms["Shapley"]["fold_mae"][str(s)]<ms["A"]["fold_mae"][str(s)] for s in folds);losses=len(folds)-wins
    hard_wins=sum(ms["Hard"]["fold_mae"][str(s)]<ms["A"]["fold_mae"][str(s)] for s in folds)
    tierout={k:{"value":tier_value[k],"n":tier_n[k],"share_of_total":tier_value[k]/system_total if system_total else 0.0} for k in sorted(tier_value)}
    return {"rows":out_rows,"metrics":ms,"fold_wins_vs_A":wins,"fold_losses_vs_A":losses,
            "hard_fold_wins_vs_A":hard_wins,"hard_fold_losses_vs_A":len(folds)-hard_wins,
            "season_diagnostics":season_diag,"low_end_allocation":tierout,
            "tier_definition":"within-season full-supply Y1 Shapley rank bands: bottom20 near_zero; 20-50 depth; 50-80 starter; 80-95 premium; top5 elite; diagnostic only",
            "hard_control_zero_share_of_full_supply":hard_zero_n/hard_n if hard_n else 0.0,
            "runtime_seconds":time.time()-runtime_start,"holdouts":len(out_rows),"folds":len(folds)}
