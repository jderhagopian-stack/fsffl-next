        return out
    return corr(ranks(xs),ranks(ys))


def subset_caps(team_count=TEAM_COUNT,direct=DIRECT,flex=FLEX,superflex=SUPERFLEX):
    caps=[]
    for mask in range(1,1<<4):
        cap=team_count*sum(direct[POSITIONS[i]] for i in range(4) if mask&(1<<i))
        if any(mask&(1<<POS_INDEX[p]) for p in ("RB","WR","TE")):
            cap += team_count*flex
        cap += team_count*superflex
        idxs=tuple(i for i in range(4) if mask&(1<<i))
        caps.append((idxs,cap))
    return tuple(caps)


class Basis:
    def __init__(self,caps):
        self.caps=caps
        self.counts=[0,0,0,0]
        # Within a position, players are interchangeable for feasibility. Keeping
        # four sorted lists makes the fundamental-circuit swap O(positions) rather
        # than scanning the whole selected basis.
        self.by_pos=[[] for _ in range(4)]  # ascending (weight, pid)
        self.total=0.0

    def _ok_counts(self,counts):
        return all(sum(counts[i] for i in idxs)<=cap for idxs,cap in self.caps)

    def _can_add(self,posi):
        c=self.counts.copy();c[posi]+=1
        return self._ok_counts(c)

    def _can_swap(self,out_pos,in_pos):
        if self.counts[out_pos] <= 0:
            return False
        c=self.counts.copy();c[out_pos]-=1;c[in_pos]+=1
        return self._ok_counts(c)

    def marginal(self,pos:str,weight:float):
        w=max(0.0,float(weight));pi=POS_INDEX[pos]
        if w<=0:return 0.0,None
        if self._can_add(pi):return w,(None,None,None)
        # In a weighted matroid basis, adding a dependent element can only improve
        # the optimum by displacing the lightest member of its fundamental circuit.
        # Position-equivalence means the lightest selected member of each feasible
        # outgoing position is sufficient to identify that candidate.
        best=None
        for opi in range(4):
            if not self.by_pos[opi] or not self._can_swap(opi,pi):
                continue
            ow,opid=self.by_pos[opi][0]
            if best is None or (ow,opid,opi) < best:
                best=(ow,opid,opi)
        if best is None or w<=best[0]+1e-12:
            return 0.0,None
        ow,opid,opi=best
        return w-ow,(opi,ow,opid)

    def add(self,pid:str,pos:str,weight:float):
        delta,replace=self.marginal(pos,weight)
        w=max(0.0,float(weight));pi=POS_INDEX[pos]
        if replace is None:
            return 0.0
        if replace[0] is None:
            bisect.insort(self.by_pos[pi],(w,pid));self.counts[pi]+=1;self.total+=w
            return w
        opi,ow,opid=replace
        # fundamental-circuit candidate is the first (lightest) selected member
        # at the outgoing position
        self.by_pos[opi].pop(0);self.counts[opi]-=1;self.total-=ow
        bisect.insort(self.by_pos[pi],(w,pid));self.counts[pi]+=1;self.total+=w
        return w-ow
