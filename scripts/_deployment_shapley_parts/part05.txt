def metrics(rows,key):
    errs=[abs(r[key]-r["target"]) for r in rows]
    targets=[r["target"] for r in rows]
    q90,q25,q50=quantile(targets,.90),quantile(targets,.25),quantile(targets,.50)
    def age_band(pos,age):
        if age is None:return "unknown"
        if pos=="QB":return "young" if age<=25 else ("prime" if age<=31 else "aging")
        return "young" if age<=23 else ("prime" if age<=27 else "aging")
    groups={
      "young":[r for r in rows if age_band(r["position"],r["age"])=="young"],
      "prime":[r for r in rows if age_band(r["position"],r["age"])=="prime"],
      "aging":[r for r in rows if age_band(r["position"],r["age"])=="aging"],
      "elite":[r for r in rows if r["target"]>=q90],
      "fringe":[r for r in rows if r["target"]<=q25],
      "developmental":[r for r in rows if age_band(r["position"],r["age"])=="young" and r["target"]<q50],
    }
    folds=defaultdict(list);pos=defaultdict(list)
    for r in rows:folds[r["season"]].append(r);pos[r["position"]].append(r)
    return {"n":len(rows),"mae":mean(errs),"correlation":corr([r[key] for r in rows],targets),
            "position_mae":{p:mean(abs(r[key]-r["target"]) for r in rs) for p,rs in pos.items()},
            "group_mae":{g:mean(abs(r[key]-r["target"]) for r in rs) for g,rs in groups.items() if rs},
            "fold_mae":{str(s):mean(abs(r[key]-r["target"]) for r in rs) for s,rs in sorted(folds.items())}}
