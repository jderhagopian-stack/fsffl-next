        legacy=load_module(here/"run_fundamental_intrinsic_residual_calibration.py","current_shapley_legacy")
        explicit=load_module(here/"run_intrinsic_explicit_state_challenge.py","current_shapley_explicit")
        panel=legacy.load_rows(args.career_panel);boundaries=explicit.fit_state_boundaries(panel,2026)
        caps=subset_caps();phi_by_h={}
        for h in (0,1,2):
            pool=[(pid,e.position.value,max(0.0,e.horizons[h].forecast_mean)) for pid,e in eby.items() if len(e.horizons)>h]
            scenario={pid:[w] for pid,pos,w in pool}
            for pid,e in named:
                mu=e.horizons[h].forecast_mean
                if h==0:scenario[pid]=[max(0.0,mu)]
                else:scenario[pid]=[max(0.0,mu)]+[projected_central(mu,s,boundaries[e.position.value]) for s in STATE_NAMES]
            est,_,_,_=mc_scenarios(pool,scenario,caps,perms=MC_PERMUTATIONS,seed=MC_SEED+900+h);phi_by_h[h]=est
        rows=[]
        for pid,e in named:
            name=pby[pid].full_name;f=sanity[name]
            score=phi_by_h[0][pid][0]
            for h,key in [(1,"y2_state_probs"),(2,"y3_state_probs")]:
                pr=f.get(key,{})
                score+=(DISCOUNT**h)*sum(float(pr.get(s,0))*phi_by_h[h][pid][1+STATE_NAMES.index(s)] for s in STATE_NAMES)
            rows.append({"player":name,"position":e.position.value,"deployment_shapley":score,"means":[hz.forecast_mean for hz in e.horizons],"state_probability_source":"frozen latest_intrinsic_result.json"})
        rows.sort(key=lambda r:(-r["deployment_shapley"],r["player"]))
        return {"available":True,"evaluation_as_of":live.evaluation_as_of.isoformat(),"sources":list(live.successful_source_ids),"pool_n":len(eby),"rows":rows}
    except Exception as exc:
        return {"available":False,"reason":f"{type(exc).__name__}: {exc}"}
