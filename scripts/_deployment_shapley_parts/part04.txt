
def fixture_validation():
    results={}
    caps=subset_caps(team_count=1,direct={"QB":1,"RB":0,"WR":0,"TE":0},flex=0,superflex=0)
    players=[("a","QB",10.0),("b","QB",6.0),("c","QB",0.0)]
    exact=exact_shapley(players,caps)
    results["single_slot_exact"]={"phi":exact,"efficiency":sum(exact.values()),"game":full_game_value(players,caps)}
    results["efficiency_pass"]=abs(sum(exact.values())-full_game_value(players,caps))<1e-9
    # symmetry
    symplayers=[("a","QB",8.0),("b","QB",8.0)]
    sym=exact_shapley(symplayers,caps)
    results["symmetry_pass"]=abs(sym["a"]-sym["b"])<1e-9
    results["dummy_pass"]=abs(exact["c"])<1e-9
    # additivity: two one-slot games, same player set, sum characteristic functions.
    p=[("a","QB",10.0),("b","QB",4.0)]
    phi1=exact_shapley(p,caps)
    p2=[("a","QB",3.0),("b","QB",9.0)]
    phi2=exact_shapley(p2,caps)
    # exact Shapley of W1+W2 by permutation
    sums={"a":0.0,"b":0.0};n=0
    for perm in permutations(["a","b"]):
        s=[];v1=v2=0.0
        for pid in perm:
            before1=full_game_value([(x,dict(p)[x] if False else "QB",0) for x in []],caps) if False else 0
            s.append(pid)
            set1=[row for row in p if row[0] in s];set2=[row for row in p2 if row[0] in s]
            nv1=full_game_value(set1,caps);nv2=full_game_value(set2,caps)
            sums[pid]+=(nv1-v1)+(nv2-v2);v1,v2=nv1,nv2
        n+=1
    phisum={k:v/n for k,v in sums.items()}
    results["additivity_pass"]=all(abs(phisum[k]-(phi1[k]+phi2[k]))<1e-9 for k in phisum)
    # MC convergence to exact small case
    scenarios={pid:[w] for pid,_,w in players}
    mc,se,cp,diag=mc_scenarios(players,scenarios,caps,perms=2048,seed=777)
    results["mc_exact_max_abs_error"]=max(abs(mc[pid][0]-exact[pid]) for pid in exact)
    results["mc_convergence_pass"]=results["mc_exact_max_abs_error"]<0.2
    # solver vs brute-force assignment intuition fixture: one RB direct + flex + SF mixed.
    caps2=subset_caps(team_count=1,direct={"QB":1,"RB":1,"WR":0,"TE":0},flex=1,superflex=1)
    mix=[("q1","QB",20),("q2","QB",15),("r1","RB",18),("r2","RB",17),("w1","WR",16),("t1","TE",2)]
    results["mixed_fixture_value"]=full_game_value(mix,caps2)
    results["mixed_fixture_expected"]=20+18+17+16
    results["mixed_fixture_pass"]=abs(results["mixed_fixture_value"]-71)<1e-9
    return results
